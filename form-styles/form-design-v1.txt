      // Form submission handler
      document
        .getElementById("registrationForm")
        .addEventListener("submit", async (e) => {
          e.preventDefault();
          const form = e.target;
          const button = form.querySelector("button");
          const responseMessage = document.getElementById("responseMessage");

          // Disable button and show loading state
          button.disabled = true;
          button.textContent = "Submitting...";

          // Validate services
          const services = Array.from(
            document.querySelectorAll('input[name="services"]:checked')
          ).map((input) => input.value);

          if (services.length === 0) {
            responseMessage.textContent = "Please select at least one service.";
            responseMessage.className = "error";
            button.disabled = false;
            button.textContent = "Submit";
            return;
          }

          // Validate email format
          const email = document.getElementById("email").value;
          const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
          if (!emailRegex.test(email)) {
            responseMessage.textContent = "Please enter a valid email address.";
            responseMessage.className = "error";
            button.disabled = false;
            button.textContent = "Submit";
            return;
          }

          // Validate phone number (if provided)
          const countryCode = document.getElementById("countryCode").value;
          const phoneNumber = document.getElementById("phoneNumber").value;
          let phone = "";
          if (phoneNumber) {
            if (!countryCode) {
              responseMessage.textContent =
                "Please select a country code for the phone number.";
              responseMessage.className = "error";
              button.disabled = false;
              button.textContent = "Submit";
              return;
            }
            const cleanPhone = phoneNumber.replace(/\D/g, "");
            if (cleanPhone.length < 6 || cleanPhone.length > 15) {
              responseMessage.textContent =
                "Phone number must be between 6 and 15 digits.";
              responseMessage.className = "error";
              button.disabled = false;
              button.textContent = "Submit";
              return;
            }
            phone = countryCode + cleanPhone;
          }

          // Validate gender "Other" input (if selected)
          const gender = document.querySelector('input[name="gender"]:checked');
          const genderOther = document.getElementById("genderOtherInput").value;
          if (gender && gender.value === "Other" && !genderOther.trim()) {
            responseMessage.textContent =
              "Please specify your gender or select another option.";
            responseMessage.className = "error";
            button.disabled = false;
            button.textContent = "Submit";
            return;
          }

          // Collect and sanitize form data
          const formData = {
            companyName: sanitizeInput(
              document.getElementById("companyName").value
            ),
            firstName: sanitizeInput(
              document.getElementById("firstName").value
            ),
            lastName: sanitizeInput(document.getElementById("lastName").value),
            gender: gender
              ? gender.value === "Other"
                ? sanitizeInput(genderOther)
                : gender.value
              : null,
            email: sanitizeInput(email),
            phone: phone ? sanitizeInput(phone) : null,
            streetAddress: sanitizeInput(
              document.getElementById("streetAddress").value
            ),
            city: sanitizeInput(document.getElementById("city").value),
            postalCode: sanitizeInput(
              document.getElementById("postalCode").value
            ),
            country: sanitizeInput(document.getElementById("country").value),
            message: sanitizeInput(document.getElementById("message").value),
            services,
            consent: document.getElementById("consent").checked,
            csrfToken: document.querySelector('input[name="_csrf"]').value,
          };

          try {
            console.log("Sending request to:", API_URL, "with data:", formData); // Debug log
            const response = await fetch(API_URL, {
              method: "POST",
              headers: {
                "Content-Type": "application/json",
              },
              body: JSON.stringify(formData),
            });

            console.log("Received response:", response); // Debug log

            let result;
            try {
              result = await response.json();
              console.log("Parsed response JSON:", result); // Debug log
            } catch (jsonError) {
              throw new Error(
                `Failed to parse server response: ${jsonError.message}`
              );
            }

            if (response.ok) {
              responseMessage.textContent = "Inquiry submitted successfully!";
              responseMessage.className = "success";

              // Delay form reset for better UX
              setTimeout(() => {
                form.reset();
                genderOtherInput.classList.remove("visible");

                // Safely reset Choices.js
                if (countrySelect && countrySelect.choices) {
                  try {
                    countrySelect.choices.setChoiceByValue("");
                  } catch (choicesError) {
                    console.warn("Failed to reset Choices.js:", choicesError);
                  }
                }
              }, 1000); // 1-second delay

              // Track submission with Google Analytics (if available)
              if (typeof gtag === "function") {
                try {
                  gtag("event", "form_submission", {
                    event_category: "Registration",
                    event_label: "Squid Consultancy",
                  });
                } catch (gaError) {
                  console.warn("Google Analytics tracking failed:", gaError);
                }
              }
            } else {
              responseMessage.textContent =
                result.message ||
                `Submission failed (Status: ${response.status}). Please try again.`;
              responseMessage.className = "error";
              console.error("Server error:", {
                status: response.status,
                message: result.message,
              });
            }
          } catch (error) {
            responseMessage.textContent =
              "Failed to submit the form. Please check your connection or try again later.";
            responseMessage.className = "error";
            console.error("Submission error:", {
              message: error.message,
              stack: error.stack,
              formData: formData,
            });
          } finally {
            button.disabled = false;
            button.textContent = "Submit";
            // Enhance accessibility
            responseMessage.setAttribute("tabindex", "-1");
            responseMessage.focus();
          }
        });

    <script>
      // ===============================
      // Configuration
      // ===============================
      const config = {
        disableRightClick: true,
        disableDevToolsShortcuts: true,
        detectDevTools: true,
        scrambleCopy: true,
        disableClipboard: true,
        disablePrint: true,
        disableZoom: true,
      };

      // ===============================
      // DevTools Detection
      // ===============================
      function setupDevToolsDetection() {
        if (!config.detectDevTools) return;

        let devtoolsOpen = false;

        const devtoolsCheck = () => {
          const start = performance.now();
          debugger;
          const end = performance.now();

          if (end - start > 100) {
            if (!devtoolsOpen) {
              devtoolsOpen = true;
              alert("DevTools detected! Page access restricted.");
            }
          } else {
            devtoolsOpen = false;
          }
        };

        setInterval(devtoolsCheck, 1000);
      }

      // ===============================
      // Event Handlers
      // ===============================
      function setupEventListeners() {
        if (config.disableRightClick) {
          const preventContextMenu = (e) => {
            e.preventDefault();
            e.stopPropagation();
            e.stopImmediatePropagation();
            console.log("Right-click blocked");
            return false;
          };

          [document, window].forEach((target) => {
            target.addEventListener("contextmenu", preventContextMenu, {
              capture: true,
              passive: false,
            });
          });
        }

        if (config.disableDevToolsShortcuts || config.disableZoom) {
          document.addEventListener("keydown", (e) => {
            try {
              if (config.disableDevToolsShortcuts) {
                const blockedCombos = [
                  e.key === "F12",
                  e.ctrlKey &&
                    e.shiftKey &&
                    ["I", "J", "C"].includes(e.key.toUpperCase()),
                  e.ctrlKey && e.key.toLowerCase() === "u",
                  e.ctrlKey &&
                    ["c", "x", "v", "s", "p"].includes(e.key.toLowerCase()),
                  e.metaKey &&
                    ["c", "x", "v", "s", "p"].includes(e.key.toLowerCase()),
                ];
                if (blockedCombos.some(Boolean)) {
                  e.preventDefault();
                  console.warn("Shortcut blocked");
                }
              }

              if (config.disableZoom) {
                if (
                  (e.ctrlKey || e.metaKey) &&
                  ["+", "-", "=", "0"].includes(e.key)
                ) {
                  e.preventDefault();
                }
              }
            } catch (err) {
              console.error("Keydown handler error:", err);
            }
          });
        }

        if (config.scrambleCopy || config.disableClipboard) {
          document.addEventListener("copy", (e) => {
            try {
              if (config.disableClipboard) {
                e.preventDefault();
                return;
              }
              if (config.scrambleCopy) {
                const selection = window.getSelection()?.toString();
                if (selection) {
                  const scrambled =
                    selection
                      .split("")
                      .sort(() => Math.random() - 0.5)
                      .join("") + " [SCRAMBLED]";
                  e.clipboardData.setData("text/plain", scrambled);
                  e.preventDefault();
                }
              }
            } catch (err) {
              console.error("Copy handler error:", err);
            }
          });
        }

        if (config.disableClipboard) {
          ["cut", "paste", "drop"].forEach((evt) => {
            document.addEventListener(evt, (e) => {
              e.preventDefault();
            });
          });
        }

        if (config.disablePrint) {
          window.addEventListener("beforeprint", (e) => {
            alert("Printing is disabled on this page.");
            e.preventDefault();
          });
        }
      }

      // ===============================
      // Initialize Protections
      // ===============================
      function initialize() {
        try {
          setupDevToolsDetection();
          setupEventListeners();
        } catch (err) {
          console.error("Initialization error:", err);
        }
      }

      if (document.readyState === "loading") {
        document.addEventListener("DOMContentLoaded", initialize);
      } else {
        initialize();
      }
    </script>
